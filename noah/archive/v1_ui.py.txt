def get(data,path):
    for key in path: data=data[key]
    return data

def put(data,path,value):
    for key in path[:-1]: data=data[key]
    data[path[-1]]=value

class ScrollPage(ttk.Frame):
    def __init__(self,parent):
        super().__init__(parent)
        self.canvas=tk.Canvas(self,highlightthickness=0,bg='#f4f6f8')
        bar=ttk.Scrollbar(self,orient='vertical',command=self.canvas.yview)
        self.content=ttk.Frame(self.canvas,padding=20)
        win=self.canvas.create_window((0,0),window=self.content,anchor='nw')
        self.content.bind('<Configure>',lambda e:self.canvas.configure(scrollregion=self.canvas.bbox('all')))
        self.canvas.bind('<Configure>',lambda e:self.canvas.itemconfigure(win,width=e.width))
        self.canvas.configure(yscrollcommand=bar.set)
        self.canvas.pack(side='left',fill='both',expand=True); bar.pack(side='right',fill='y')
        self.bind('<Enter>',lambda e:self.canvas.bind_all('<MouseWheel>',self.wheel))
        self.bind('<Leave>',lambda e:self.canvas.unbind_all('<MouseWheel>'))
    def wheel(self,event):
        self.canvas.yview_scroll(-int(event.delta/120),'units')

    def box(self,parent,title):
        b=ttk.LabelFrame(parent,text=title); b.pack(fill='x',pady=(0,16)); return b

    def field(self,parent,label,path,row,col=0,choices=None,wide=False):
        ttk.Label(parent,text=label).grid(row=row,column=col*2,sticky='nw',padx=(4,10),pady=5)
        value=get(self.card,path)
        var=tk.BooleanVar() if type(value) is bool else tk.StringVar()
        if type(value) is bool: widget=ttk.Checkbutton(parent,variable=var,command=self.changed)
        elif wide:
            widget=tk.Text(parent,height=4,wrap='word',font=('Microsoft YaHei UI',10),relief='solid',bd=1,undo=True)
            widget.bind('<<Modified>>',lambda e,w=widget:self.text_changed(w))
        elif choices:
            widget=ttk.Combobox(parent,textvariable=var,values=choices,state='readonly',width=26)
        else: widget=ttk.Entry(parent,textvariable=var,width=28)
        widget.grid(row=row,column=col*2+1,sticky='ew',padx=4,pady=5); parent.columnconfigure(col*2+1,weight=1)
        self.fields[tuple(path)]=(var,widget,type(value))
        if not wide: var.trace_add('write',lambda *args:self.changed())
        return widget

    def text_changed(self,widget):
        if widget.edit_modified():
            widget.edit_modified(False); self.changed()

    def changed(self):
        if self.loading: return
        self.dirty=True; self.title('● NOAH / 诺阿人物卡')
        if self.pending: self.after_cancel(self.pending)
        self.pending=self.after(180,self.refresh)

    def collect(self):
        candidate=copy.deepcopy(self.card)
        for path,(var,widget,kind) in self.fields.items():
            if isinstance(widget,tk.Text): value=widget.get('1.0','end-1c')
            elif kind is bool: value=var.get()
            elif kind in (int,float):
                try: value=float(var.get())
                except ValueError: raise CardError(f'{" / ".join(path)} 需要有效数字')
            else: value=var.get()
            put(candidate,path,value)
        if candidate['race']!=self.card['race']: candidate['dice']={k:[] for k in ATTRS}
        validate(candidate); return candidate

    def refresh(self):
        self.pending=None
        try:
            self.card=self.collect(); self.render()
        except CardError as exc: self.status.configure(text='输入尚未有效，计算结果保留上次值：'+str(exc),foreground='#a33e27')

    def populate(self):
        self.loading=True
        for path,(var,widget,kind) in self.fields.items():
            value=get(self.card,path)
            if isinstance(widget,tk.Text): widget.delete('1.0','end'); widget.insert('1.0',value); widget.edit_modified(False)
            else: var.set(value if kind is bool else (f'{value:g}' if kind in (int,float) else value))
        self.loading=False; self.render()

    def render(self):
        values=calculate(self.card); race=CATALOG['races'][self.card['race']]
        self.file_label.configure(text=str(self.path) if self.path else '未保存的人物卡  ·  Ctrl+S 保存  /  Ctrl+O 打开')
        self.race_label.configure(text=f"种族：{race['group']}  ·  起源：{race['origin']}")
        self.race_details.configure(text=race['description']+'\n种族奖励骰：'+race['dice']+'\n战斗职业：'+CATALOG['careers'].get(self.card['basic']['战斗职业'],'未选择'))
        for k,l in self.body_labels.items(): l.configure(text=f"上限 {values['maxima'][k]:.2f}    恢复 {values['regen'][k]:+.2f} / t")
        self.summary.configure(text='     '.join(f"{k}  {v['最终']:g}" for k,v in values['attributes'].items())+f"     幸运  {self.card['body']['幸运']:g}")
        self.attr_tree.delete(*self.attr_tree.get_children())
        for k,v in values['attributes'].items(): self.attr_tree.insert('', 'end',values=[k]+[f'{v[c]:g}' for c in self.attr_tree['columns'][1:]])
        self.dice_label.configure(text=f"规则：{race['dice']}  |  记录："+'；'.join(k+':'+','.join(map(str,v)) for k,v in self.card['dice'].items() if v))
        self.level_label.configure(text=f"综合灵能等级  {values['level']:.2f} / 100   ·   {values['grade']} 级（暂定公式）")
        self.skill_tree.delete(*self.skill_tree.get_children())
        for k,v in self.card['skills'].items(): self.skill_tree.insert('', 'end',iid=k,values=[k,v['基础'],v['成长'],v['基础']+v['成长'],' / '.join(f'{a}{n:+g}' for a,n in v['效果'].items() if n)])
        self.draw_inventory(); self.weight_label.configure(text=f"总重量 {values['weight']:g} kg")
        self.status.configure(text='；'.join(values['warnings']) or '所有输入有效 · 已自动更新计算结果',foreground='#a33e27' if values['warnings'] else '#007477')

    def draw_inventory(self):
        self.bag_size.set(' × '.join(map(str,self.card['inventory']['size'])))
        for name,canvas in self.grids.items():
            w,h=self.card['inventory']['size'] if name=='背包' else CONTAINERS[name]; unit=34
            canvas.configure(width=w*unit,height=h*unit); canvas.delete('all')
            for x in range(w+1): canvas.create_line(x*unit,0,x*unit,h*unit,fill='#c3cfd8')
            for y in range(h+1): canvas.create_line(0,y*unit,w*unit,y*unit,fill='#c3cfd8')
            for item in self.card['inventory']['items']:
                if item['container']!=name: continue
                x,y=item['x']*unit,item['y']*unit; ww,hh=item['w']*unit,item['h']*unit
                canvas.create_rectangle(x+2,y+2,x+ww-2,y+hh-2,fill='#3c737e',outline='#24535d')
                canvas.create_text(x+ww/2,y+hh/2,text=item['name'],width=max(20,ww-4),fill='white',font=('Microsoft YaHei UI',9))
        equipped={i['slot']:i for i in self.card['inventory']['items'] if i['container']=='装备'}
        for slot,label in self.equipment_labels.items():
            locked=slot in SLOTS[1:5] and '赋能笔' not in equipped
            label.configure(text=slot+'  /  '+('需先装备赋能笔' if locked else equipped.get(slot,{}).get('name','空')),foreground='#9299a0' if locked else '#243347')
        self.item_tree.delete(*self.item_tree.get_children())
        for i in self.card['inventory']['items']: self.item_tree.insert('', 'end',iid=i['id'],values=[i['name'],i['container']+(' / '+i['slot'] if i['slot'] else ''),f"{i['w']}×{i['h']} @ {i['x']},{i['y']}",i['quantity'],i['weight'],i['description']])

    def roll(self):
        try: candidate=self.collect()
        except CardError as exc: return messagebox.showerror('输入错误',str(exc))
        if any(candidate['dice'].values()) and not messagebox.askyesno('重投奖励骰','重投将替换已保存的骰点，是否继续？'): return
        roll_race(candidate); self.card=candidate; self.changed(); self.render()

    def resize_bag(self,event=None):
        try:
            candidate=self.collect(); candidate['inventory']['size']=[int(v.strip()) for v in self.bag_size.get().split('×')]; validate(candidate)
            self.card=candidate; self.changed(); self.render()
        except CardError as exc: self.bag_size.set(' × '.join(map(str,self.card['inventory']['size']))); messagebox.showerror('无法改变容量',str(exc))

    def grid_click(self,name,event):
        x,y=event.x//34,event.y//34
        for i in self.card['inventory']['items']:
            if i['container']==name and i['x']<=x<i['x']+i['w'] and i['y']<=y<i['y']+i['h']: return self.edit_item(i['id'])
        self.edit_item(container=name,x=x,y=y)

    def equip_click(self,slot):
        items=self.card['inventory']['items']; has_pen=any(i['container']=='装备' and i['slot']=='赋能笔' for i in items)
        if slot in SLOTS[1:5] and not has_pen: return messagebox.showinfo('部件尚未解锁','请先在赋能笔槽装备赋能笔。')
        for i in items:
            if i['container']=='装备' and i['slot']==slot: return self.edit_item(i['id'])
        self.edit_item(container='装备',slot=slot)

    def edit_selected_item(self):
        selection=self.item_tree.selection()
        if selection: self.edit_item(selection[0])

    def delete_item(self):
        selection=self.item_tree.selection()
        if not selection: return
        if not messagebox.askyesno('删除物品','删除所选物品？删除赋能笔前，请先将其部件移回背包。'): return
        try:
            candidate=self.collect(); candidate['inventory']['items']=[i for i in candidate['inventory']['items'] if i['id']!=selection[0]]; validate(candidate)
            self.card=candidate; self.changed(); self.render()
        except CardError as exc: messagebox.showerror('无法删除',str(exc))

    def edit_item(self,item_id=None,container='背包',x=0,y=0,slot=''):
        try: candidate=self.collect()
        except CardError as exc: return messagebox.showerror('输入错误',str(exc))
        old=next((i for i in candidate['inventory']['items'] if i['id']==item_id),None)
        item=copy.deepcopy(old) if old else {'id':uuid.uuid4().hex,'name':'','container':container,'x':x,'y':y,'w':1,'h':1,'quantity':1,'weight':0.0,'slot':slot,'effects':{k:0.0 for k in EFFECTS},'description':''}
        win,body=self.dialog('物品 · 尺寸 / 位置 / 装备效果'); variables={}
        options={'container':list(CONTAINERS)+['装备'],'slot':['']+SLOTS}
        labels={'name':'名称','container':'所在容器','slot':'装备槽','x':'横坐标 x','y':'纵坐标 y','w':'宽度','h':'高度','quantity':'数量','weight':'单件重量 kg','description':'功能说明'}
        for r,(k,label) in enumerate(labels.items()):
            ttk.Label(body,text=label).grid(row=r,column=0,sticky='w',pady=4)
            var=tk.StringVar(value=str(item[k])); variables[k]=var
            if k in options: widget=ttk.Combobox(body,textvariable=var,values=options[k],state='readonly')
            else: widget=ttk.Entry(body,textvariable=var,width=50)
            widget.grid(row=r,column=1,sticky='ew',pady=4)
        body.columnconfigure(1,weight=1)
        def rotate():
            width,height=variables['w'].get(),variables['h'].get(); variables['w'].set(height); variables['h'].set(width)
        ttk.Button(body,text='旋转占格（交换宽高）',command=rotate).grid(row=10,column=1,sticky='w',pady=5)
        ttk.Label(body,text='效果仅在装备时生效；每件装备只计算一次。').grid(row=11,column=0,columnspan=2,sticky='w')
        for r,k in enumerate(EFFECTS,12):
            ttk.Label(body,text=k).grid(row=r,column=0,sticky='w',pady=3); var=tk.StringVar(value=f"{item['effects'][k]:g}"); variables['effect:'+k]=var; ttk.Entry(body,textvariable=var).grid(row=r,column=1,sticky='ew',pady=3)
        def apply():
            try:
                edited=copy.deepcopy(item)
                for k in labels:
                    text=variables[k].get(); edited[k]=int(text) if k in ['x','y','w','h','quantity'] else float(text) if k=='weight' else text
                if edited['container']!='装备': edited['slot']=''
                edited['effects']={k:float(variables['effect:'+k].get()) for k in EFFECTS}
                candidate['inventory']['items']=[i for i in candidate['inventory']['items'] if i['id']!=edited['id']]+[edited]
                validate(candidate); self.card=candidate; self.changed(); self.render(); win.destroy()
            except (ValueError,CardError) as exc: messagebox.showerror('物品无效',str(exc),parent=win)
        footer=ttk.Frame(win,padding=10); footer.pack(fill='x'); ttk.Button(footer,text='保存物品',command=apply).pack(side='right'); ttk.Button(footer,text='取消',command=win.destroy).pack(side='right',padx=8)

    def edit_skill(self):
        selection=self.skill_tree.selection()
        if not selection: return
        try: candidate=self.collect()
        except CardError as exc: return messagebox.showerror('输入错误',str(exc))
        key=selection[0]; skill=candidate['skills'][key]; win,body=self.dialog(key+' · 技能效果'); variables={}
        for r,k in enumerate(['基础','成长','描述']+ATTRS):
            ttk.Label(body,text=k).grid(row=r,column=0,sticky='w',pady=8)
            var=tk.StringVar(value=skill[k] if k in skill else skill['效果'][k]); variables[k]=var
            ttk.Entry(body,textvariable=var,width=50).grid(row=r,column=1,sticky='ew',pady=8)
        def apply():
            try:
                skill['基础']=float(variables['基础'].get()); skill['成长']=float(variables['成长'].get()); skill['描述']=variables['描述'].get()
                skill['效果']={k:float(variables[k].get()) for k in ATTRS}; validate(candidate)
                self.card=candidate; self.changed(); self.render(); win.destroy()
            except (ValueError,CardError) as exc: messagebox.showerror('技能无效',str(exc),parent=win)
        ttk.Button(body,text='保存技能',command=apply).grid(row=12,column=1,pady=15)

    def confirm_discard(self):
        if not self.dirty: return True
        answer=messagebox.askyesnocancel('有未保存修改','是否先保存当前人物卡？\n选“否”将放弃当前未保存修改。')
        if answer is None: return False
        return self.save() if answer else True

    def new(self):
        if not self.confirm_discard(): return
        self.card=new_card(); self.path=None; self.dirty=False; self.populate(); self.title('NOAH / 诺阿人物卡')

    def open(self):
        if not self.confirm_discard(): return
        path=filedialog.askopenfilename(filetypes=[('诺阿人物卡','*.noah *.json *.xml *.bak'),('所有文件','*.*')])
        if not path: return
        try: candidate=load(path)
        except CardError as exc: return messagebox.showerror('打开失败',str(exc))
        self.card=candidate; self.path=Path(path); self.dirty=False; self.populate(); self.title('NOAH / 诺阿人物卡')

    def save(self,as_new=False):
        try: candidate=self.collect()
        except CardError as exc: messagebox.showerror('无法保存',str(exc)); return False
        path=self.path
        if as_new or path is None:
            selected=filedialog.asksaveasfilename(defaultextension='.noah',initialfile=(candidate['basic']['姓名'] or '未命名人物')+'.noah',filetypes=[('诺阿人物卡','*.noah'),('JSON','*.json'),('XML','*.xml')])
            if not selected: return False
            path=Path(selected)
        try: save(path,candidate)
        except CardError as exc: messagebox.showerror('保存失败',str(exc)); return False
        self.card=candidate; self.path=path; self.dirty=False; self.title('NOAH / 诺阿人物卡'); self.render(); self.status.configure(text='已保存：'+str(path),foreground='#007477'); return True

    def export_xml(self):
        try: candidate=self.collect()
        except CardError as exc: return messagebox.showerror('无法导出',str(exc))
        path=filedialog.asksaveasfilename(defaultextension='.xml',filetypes=[('XML','*.xml')],initialfile='人物卡.xml')
        if path:
            try: save(path,candidate); self.status.configure(text='XML 已导出，可通过“打开”继续编辑：'+path)
            except CardError as exc: messagebox.showerror('导出失败',str(exc))

    def autosave(self):
        if self.dirty:
            try:
                candidate=self.collect(); self.recovery.parent.mkdir(exist_ok=True); save(self.recovery,candidate)
            except (CardError,OSError) as exc: self.status.configure(text='草稿未能保存：'+str(exc),foreground='#a33e27')
        self.after(30000,self.autosave)

    def restore(self):
        if not self.recovery.exists(): return messagebox.showinfo('恢复草稿','暂无自动保存草稿。编辑期间每30秒保存一次有效数据。')
        if not self.confirm_discard(): return
        try: self.card=load(self.recovery)
        except CardError as exc: return messagebox.showerror('草稿无法恢复',str(exc))
        self.path=None; self.dirty=True; self.populate(); self.title('● NOAH / 恢复的草稿')

    def close(self):
        if self.confirm_discard(): self.destroy()

    def callback_error(self,exc_type,exc,tb):
        logging.error('Unexpected UI error',exc_info=(exc_type,exc,tb))
        messagebox.showerror('操作未完成','发生意外错误。当前窗口仍可使用；请尝试另存为。错误类型：'+exc_type.__name__+'\n详情见 .local/app.log（不包含人物卡内容）。')
